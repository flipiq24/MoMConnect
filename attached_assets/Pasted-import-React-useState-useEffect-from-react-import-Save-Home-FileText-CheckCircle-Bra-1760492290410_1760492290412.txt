import React, { useState, useEffect } from 'react';
import { Save, Home, FileText, CheckCircle, Brain, AlertCircle, TrendingUp, Calculator } from 'lucide-react';

const MoMWholesaleSystem = () => {
  const [currentTab, setCurrentTab] = useState('login');
  const [userData, setUserData] = useState({ name: '', email: '' });
  const [propertyData, setPropertyData] = useState({});
  const [autoSaving, setAutoSaving] = useState(false);
  const [lastSaved, setLastSaved] = useState(null);
  const [zillowData, setZillowData] = useState(null);
  const [aiAnalysis, setAiAnalysis] = useState(null);
  const [riskScore, setRiskScore] = useState(0);
  const [emdRecommendation, setEmdRecommendation] = useState(null);

  // Point system scoring rules
  const scoringRules = {
    'Days in the MLS from acceptance Date': {
      'Not In the MLS': 1,
      'Less than one week': 0,
      'PR/BM': 0,
      'More than a Week': -1
    },
    'Is purchase price over 1M': {
      '1M or under': 1,
      '1-1.5M': 0,
      '1.5M +': -1
    },
    'Is value subject to permits': {
      'No': 0,
      'Approved Permits': 0,
      'Yes': -2
    },
    'Is value subject to ADU?': {
      'No': 0,
      'Approved Permits': 0,
      'Yes': -3
    },
    'Is wholesale price higher than listing asking price?': {
      'Significantly lower (10%+ below asking)': 1,
      '5% below asking': 0,
      'Below 5% below asking': -2,
      'Over asking': -3
    },
    'ARV Confidence': {
      'Sold Comps': 1,
      'little to no Comps': 0,
      'Pushing Value': -1
    },
    'ROI for time/Effort': {
      'High': 1,
      'Standard': 0,
      'Low': -1
    },
    'Zoning': {
      'Residential': 0,
      'Commercial with 100% Burn Letter': -1,
      'Commercial (Legal Non Conforming)': -3
    },
    'Rehab Level': {
      'Light': 1,
      'Medium': 0,
      'Heavy': -1
    },
    'Area Desirability': {
      'High': 1,
      'Standard': 0,
      'Low': -1
    },
    'Obsolesces/Issues': {
      'No': 0,
      'Minor': -1,
      'Modest': -2,
      'Major': -3
    },
    'Occupancy': {
      'Vacant at COE': 1,
      'Seller/Tenant will hold back funds and keep occupancy': -1,
      'Seller Will stay for more than 3 months no hold back': -2,
      'Tenant occupied': -3
    }
  };

  // EXACT EMD Decision Matrix from user specifications
  const getEMDRecommendation = (score) => {
    const matrix = {
      5: { emd: 'Yes EMD', chance: 100 },
      4: { emd: 'Yes EMD', chance: 90 },
      3: { emd: 'Yes EMD', chance: 80 },
      2: { emd: 'Yes EMD', chance: 70 },
      1: { emd: 'Yes EMD', chance: 60 },
      0: { emd: 'Yes EMD', chance: 50 },
      '-1': { emd: 'TBD', chance: 40 },
      '-2': { emd: 'No EMD', chance: 30 },
      '-3': { emd: 'No EMD', chance: 20 },
      '-4': { emd: 'No EMD', chance: 10 },
      '-5': { emd: 'No EMD', chance: 0 }
    };
    
    // Handle scores outside the matrix
    if (score > 5) return { emd: 'Yes EMD', chance: 100 };
    if (score < -5) return { emd: 'No EMD', chance: 0 };
    
    return matrix[score.toString()] || { emd: 'TBD', chance: 40 };
  };

  // Get color for point value
  const getPointColor = (points) => {
    if (points > 0) return 'bg-green-100 text-green-800 border-green-300';
    if (points < 0) return 'bg-red-100 text-red-800 border-red-300';
    return 'bg-white text-gray-800 border-gray-300';
  };

  // Get color for select options
  const getSelectColor = (field, value) => {
    if (!value || !scoringRules[field]) return '';
    const points = scoringRules[field][value];
    if (points > 0) return 'bg-green-50 border-green-500';
    if (points < 0) return 'bg-red-50 border-red-500';
    return 'bg-white border-gray-300';
  };

  // Get EMD status color
  const getEMDStatusColor = (emdStatus) => {
    if (emdStatus === 'Yes EMD') return 'bg-green-500 text-white';
    if (emdStatus === 'TBD') return 'bg-yellow-500 text-white';
    if (emdStatus === 'No EMD') return 'bg-red-500 text-white';
    return 'bg-gray-500 text-white';
  };

  // Calculate risk score
  useEffect(() => {
    let totalScore = 0;
    Object.keys(scoringRules).forEach(field => {
      const value = propertyData[field];
      if (value && scoringRules[field][value] !== undefined) {
        totalScore += scoringRules[field][value];
      }
    });
    setRiskScore(totalScore);
    setEmdRecommendation(getEMDRecommendation(totalScore));
  }, [propertyData]);

  // Auto-save to Google Sheets
  useEffect(() => {
    if (propertyData.address && Object.keys(propertyData).length > 1) {
      const timer = setTimeout(() => {
        saveToGoogleSheets();
      }, 2000);
      return () => clearTimeout(timer);
    }
  }, [propertyData]);

  const saveToGoogleSheets = async () => {
    setAutoSaving(true);
    try {
      await new Promise(resolve => setTimeout(resolve, 500));
      setLastSaved(new Date());
    } catch (error) {
      console.error('Save error:', error);
    }
    setAutoSaving(false);
  };

  const fetchZillowData = async (address) => {
    try {
      await new Promise(resolve => setTimeout(resolve, 1000));
      const mockData = {
        address: address,
        zpid: '12345678',
        price: 850000,
        bedrooms: 3,
        bathrooms: 2,
        sqft: 1800,
        lotSize: 7200,
        yearBuilt: 1985,
        zestimate: 875000
      };
      setZillowData(mockData);
      setPropertyData(prev => ({
        ...prev,
        'Estimated Rehab': 75000,
        'ARV': mockData.zestimate
      }));
    } catch (error) {
      console.error('Zillow fetch error:', error);
    }
  };

  const generateAIAnalysis = async () => {
    try {
      await new Promise(resolve => setTimeout(resolve, 1500));
      
      const analysis = {
        summary: `Based on the comprehensive analysis of ${propertyData.address}, this property has a total score of ${riskScore} points with a ${emdRecommendation?.chance}% estimated wholesale success rate.`,
        strengths: riskScore > 0 ? [
          'Positive risk factors outweigh concerns',
          'Property meets key criteria for wholesaling',
          'Market conditions appear favorable'
        ] : [
          'Standard risk profile',
          'Requires careful due diligence'
        ],
        concerns: riskScore < 0 ? [
          'Negative risk factors present',
          'Additional review required',
          'Consider risk mitigation strategies'
        ] : ['Minimal concerns identified'],
        recommendation: emdRecommendation?.emd === 'Yes EMD' 
          ? 'PROCEED with EMD deposit. Risk factors are within acceptable range.'
          : emdRecommendation?.emd === 'TBD'
          ? 'TO BE DETERMINED. Score at threshold - additional review and manager approval required.'
          : 'DO NOT SEND EMD. Risk score exceeds threshold. Property requires additional review or pass.',
        breakdown: Object.keys(scoringRules).map(field => ({
          field,
          value: propertyData[field] || 'Not set',
          points: scoringRules[field][propertyData[field]] || 0
        }))
      };
      
      setAiAnalysis(analysis);
    } catch (error) {
      console.error('AI analysis error:', error);
    }
  };

  const handleLogin = () => {
    if (userData.name && userData.email) {
      localStorage.setItem('momUser', JSON.stringify(userData));
      setCurrentTab('acquisition');
    }
  };

  useEffect(() => {
    const savedUser = localStorage.getItem('momUser');
    if (savedUser) {
      setUserData(JSON.parse(savedUser));
    }
  }, []);

  const updateField = (field, value) => {
    setPropertyData(prev => ({
      ...prev,
      [field]: value
    }));
  };

  // Login Screen
  if (currentTab === 'login' && !userData.name) {
    return (
      <div className="min-h-screen bg-gradient-to-br from-blue-50 to-indigo-100 flex items-center justify-center p-4">
        <div className="bg-white rounded-2xl shadow-2xl p-8 w-full max-w-md">
          <div className="text-center mb-8">
            <div className="inline-block p-3 bg-indigo-100 rounded-full mb-4">
              <Home className="w-8 h-8 text-indigo-600" />
            </div>
            <h1 className="text-3xl font-bold text-gray-900 mb-2">MoM Wholesale System</h1>
            <p className="text-gray-600">Property Management & Analysis Platform</p>
          </div>
          
          <div className="space-y-4">
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">Full Name</label>
              <input
                type="text"
                value={userData.name}
                onChange={(e) => setUserData(prev => ({ ...prev, name: e.target.value }))}
                className="w-full px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-transparent"
                placeholder="John Doe"
              />
            </div>
            
            <div>
              <label className="block text-sm font-medium text-gray-700 mb-2">Email Address</label>
              <input
                type="email"
                value={userData.email}
                onChange={(e) => setUserData(prev => ({ ...prev, email: e.target.value }))}
                className="w-full px-4 py-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500 focus:border-transparent"
                placeholder="john@company.com"
              />
            </div>
            
            <button
              onClick={handleLogin}
              disabled={!userData.name || !userData.email}
              className="w-full bg-indigo-600 text-white py-3 rounded-lg font-semibold hover:bg-indigo-700 disabled:bg-gray-300 disabled:cursor-not-allowed transition-colors"
            >
              Access System
            </button>
          </div>
        </div>
      </div>
    );
  }

  // Main Application
  return (
    <div className="min-h-screen bg-gray-50">
      {/* Header with Total Points Display */}
      <div className="bg-white shadow-sm border-b border-gray-200">
        <div className="max-w-7xl mx-auto px-4 py-4">
          <div className="flex items-center justify-between">
            <div className="flex items-center space-x-3">
              <Home className="w-6 h-6 text-indigo-600" />
              <div>
                <h1 className="text-xl font-bold text-gray-900">MoM Wholesale System</h1>
                <p className="text-sm text-gray-600">{userData.name} • {userData.email}</p>
              </div>
            </div>
            
            <div className="flex items-center space-x-4">
              {/* TOTAL POINTS DISPLAY */}
              <div className={`px-6 py-3 rounded-lg border-2 ${
                riskScore >= 1 ? 'bg-green-50 border-green-500' :
                riskScore === 0 ? 'bg-blue-50 border-blue-500' :
                riskScore === -1 ? 'bg-yellow-50 border-yellow-500' :
                'bg-red-50 border-red-500'
              }`}>
                <div className="flex items-center space-x-3">
                  <Calculator className="w-5 h-5 text-gray-700" />
                  <div>
                    <p className="text-xs text-gray-600 font-medium">Total Points</p>
                    <p className={`text-2xl font-bold ${
                      riskScore >= 1 ? 'text-green-700' :
                      riskScore === 0 ? 'text-blue-700' :
                      riskScore === -1 ? 'text-yellow-700' :
                      'text-red-700'
                    }`}>
                      {riskScore > 0 ? '+' : ''}{riskScore}
                    </p>
                  </div>
                  <div className={`px-3 py-1 rounded font-semibold text-sm ${getEMDStatusColor(emdRecommendation?.emd)}`}>
                    {emdRecommendation?.emd || 'N/A'}
                  </div>
                </div>
              </div>

              {autoSaving ? (
                <span className="flex items-center text-sm text-gray-600">
                  <Save className="w-4 h-4 mr-2 animate-pulse" />
                  Saving...
                </span>
              ) : lastSaved ? (
                <span className="text-sm text-green-600">
                  Saved {lastSaved.toLocaleTimeString()}
                </span>
              ) : null}
            </div>
          </div>
        </div>
      </div>

      {/* Navigation Tabs */}
      <div className="bg-white border-b border-gray-200">
        <div className="max-w-7xl mx-auto px-4">
          <div className="flex space-x-1">
            {[
              { id: 'acquisition', label: 'Acquisition Associate', icon: FileText },
              { id: 'mom', label: 'MOM Meeting', icon: CheckCircle },
              { id: 'approval', label: 'AM Hard Approval', icon: TrendingUp },
              { id: 'ai', label: 'AI Report', icon: Brain }
            ].map(tab => (
              <button
                key={tab.id}
                onClick={() => setCurrentTab(tab.id)}
                className={`flex items-center space-x-2 px-6 py-4 border-b-2 font-medium transition-colors ${
                  currentTab === tab.id
                    ? 'border-indigo-600 text-indigo-600'
                    : 'border-transparent text-gray-600 hover:text-gray-900 hover:border-gray-300'
                }`}
              >
                <tab.icon className="w-4 h-4" />
                <span>{tab.label}</span>
              </button>
            ))}
          </div>
        </div>
      </div>

      {/* Content Area */}
      <div className="max-w-7xl mx-auto px-4 py-6">
        {/* Acquisition Associate Tab */}
        {currentTab === 'acquisition' && (
          <div className="space-y-6">
            <div className="bg-white rounded-lg shadow-sm p-6">
              <h2 className="text-2xl font-bold text-gray-900 mb-6">Property Information</h2>
              
              {/* Property Address & Zillow Integration */}
              <div className="mb-6">
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  Property Address *
                </label>
                <div className="flex space-x-2">
                  <input
                    type="text"
                    value={propertyData.address || ''}
                    onChange={(e) => updateField('address', e.target.value)}
                    className="flex-1 px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500"
                    placeholder="123 Main Street, City, State ZIP"
                  />
                  <button
                    onClick={() => fetchZillowData(propertyData.address)}
                    disabled={!propertyData.address}
                    className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:bg-gray-300"
                  >
                    Fetch Zillow Data
                  </button>
                </div>
                {zillowData && (
                  <div className="mt-2 p-3 bg-green-50 border border-green-200 rounded-lg">
                    <p className="text-sm text-green-800">
                      ✓ Zillow data loaded: {zillowData.bedrooms}bed/{zillowData.bathrooms}bath, 
                      {zillowData.sqft}sqft, Built {zillowData.yearBuilt}
                    </p>
                  </div>
                )}
              </div>

              <div className="grid grid-cols-2 gap-6">
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    ARV (After Repair Value)
                  </label>
                  <input
                    type="number"
                    value={propertyData['ARV'] || ''}
                    onChange={(e) => updateField('ARV', e.target.value)}
                    className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500"
                    placeholder="$875,000"
                  />
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Estimated Rehab
                  </label>
                  <input
                    type="number"
                    value={propertyData['Estimated Rehab'] || ''}
                    onChange={(e) => updateField('Estimated Rehab', e.target.value)}
                    className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500"
                    placeholder="$75,000"
                  />
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Acquisition Purchase Price
                  </label>
                  <input
                    type="number"
                    value={propertyData['Acquisition purchase price'] || ''}
                    onChange={(e) => updateField('Acquisition purchase price', e.target.value)}
                    className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500"
                    placeholder="$650,000"
                  />
                  {propertyData['Acquisition purchase price'] && (
                    <p className="text-xs text-gray-500 mt-1">
                      Est. Closing Cost (0.65%): ${(parseFloat(propertyData['Acquisition purchase price']) * 0.0065).toLocaleString(undefined, {maximumFractionDigits: 0})}
                    </p>
                  )}
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Sale Price to Wholesale Buyer (Calculated)
                  </label>
                  <div className="px-4 py-2 bg-gray-100 border border-gray-300 rounded-lg">
                    <p className="text-lg font-semibold text-gray-900">
                      ${(() => {
                        const acquisitionPrice = parseFloat(propertyData['Acquisition purchase price']) || 0;
                        const wholesaleFee = parseFloat(propertyData['Forecasted Wholesale Fee']) || 0;
                        const salePrice = acquisitionPrice + wholesaleFee;
                        // Auto-update the field
                        if (salePrice > 0) {
                          propertyData['Sale Price to wholesale buyer'] = salePrice;
                        }
                        return salePrice.toLocaleString(undefined, {maximumFractionDigits: 0});
                      })()}
                    </p>
                    <p className="text-xs text-gray-500 mt-1">
                      Acquisition + Wholesale Fee
                    </p>
                  </div>
                </div>

                <div className="col-span-2">
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Forecasted Wholesale Fee
                  </label>
                  <div className="flex items-center space-x-4">
                    <input
                      type="number"
                      value={propertyData['Forecasted Wholesale Fee'] || ''}
                      onChange={(e) => updateField('Forecasted Wholesale Fee', e.target.value)}
                      className="flex-1 px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500"
                      placeholder="$50,000"
                    />
                    
                    {/* ARV Percentage Calculator */}
                    {propertyData['ARV'] && propertyData['Acquisition purchase price'] && propertyData['Estimated Rehab'] && propertyData['Forecasted Wholesale Fee'] && (
                      (() => {
                        const arv = parseFloat(propertyData['ARV']) || 0;
                        const acquisitionPrice = parseFloat(propertyData['Acquisition purchase price']) || 0;
                        const estimatedRehab = parseFloat(propertyData['Estimated Rehab']) || 0;
                        const wholesaleFee = parseFloat(propertyData['Forecasted Wholesale Fee']) || 0;
                        const closingCost = acquisitionPrice * 0.0065;
                        const totalCost = acquisitionPrice + estimatedRehab + wholesaleFee + closingCost;
                        const percentage = arv > 0 ? (totalCost / arv) * 100 : 0;
                        
                        let bgColor, textColor, borderColor, statusText;
                        if (percentage <= 83) {
                          bgColor = 'bg-green-100';
                          textColor = 'text-green-800';
                          borderColor = 'border-green-500';
                          statusText = 'Excellent Deal';
                        } else if (percentage > 83 && percentage <= 85) {
                          bgColor = 'bg-yellow-100';
                          textColor = 'text-yellow-800';
                          borderColor = 'border-yellow-500';
                          statusText = 'Caution Zone';
                        } else {
                          bgColor = 'bg-red-100';
                          textColor = 'text-red-800';
                          borderColor = 'border-red-500';
                          statusText = 'High Risk';
                        }
                        
                        return (
                          <div className={`flex-1 px-4 py-3 rounded-lg border-2 ${bgColor} ${borderColor}`}>
                            <div className="flex items-center justify-between">
                              <div>
                                <p className="text-xs text-gray-600 font-medium">Total All-In Cost / ARV</p>
                                <p className={`text-2xl font-bold ${textColor}`}>
                                  {percentage.toFixed(1)}%
                                </p>
                              </div>
                              <div className="text-right">
                                <p className={`text-xs font-semibold ${textColor} mb-1`}>{statusText}</p>
                                <p className="text-xs text-gray-600">
                                  ${totalCost.toLocaleString(undefined, {maximumFractionDigits: 0})} / ${arv.toLocaleString()}
                                </p>
                                <p className="text-xs text-gray-500 mt-1">
                                  Includes 0.65% closing cost
                                </p>
                                <p className="text-xs text-gray-500 mt-2">
                                  🟢 ≤83% | 🟡 83.1-85% | 🔴 >85%
                                </p>
                              </div>
                            </div>
                          </div>
                        );
                      })()
                    )}
                  </div>
                </div>
              </div>
            </div>

            {/* Risk Factors Section with Color Coding */}
            <div className="bg-white rounded-lg shadow-sm p-6">
              <h2 className="text-2xl font-bold text-gray-900 mb-4">Risk Assessment Factors</h2>
              <p className="text-sm text-gray-600 mb-6">
                🟢 Green = Positive points | ⚪ White = Neutral (0 points) | 🔴 Red = Negative points
              </p>
              
              <div className="grid grid-cols-2 gap-6">
                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Days in the MLS from acceptance Date
                  </label>
                  <select
                    value={propertyData['Days in the MLS from acceptance Date'] || ''}
                    onChange={(e) => updateField('Days in the MLS from acceptance Date', e.target.value)}
                    className={`w-full px-4 py-2 border-2 rounded-lg focus:ring-2 focus:ring-indigo-500 ${
                      getSelectColor('Days in the MLS from acceptance Date', propertyData['Days in the MLS from acceptance Date'])
                    }`}
                  >
                    <option value="">Select...</option>
                    <option value="Not In the MLS">Not In the MLS (+1)</option>
                    <option value="Less than one week">Less than one week (0)</option>
                    <option value="PR/BM">PR/BM (0)</option>
                    <option value="More than a Week">More than a Week (-1)</option>
                  </select>
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Is purchase price over 1M
                  </label>
                  <select
                    value={propertyData['Is purchase price over 1M'] || ''}
                    onChange={(e) => updateField('Is purchase price over 1M', e.target.value)}
                    className={`w-full px-4 py-2 border-2 rounded-lg focus:ring-2 focus:ring-indigo-500 ${
                      getSelectColor('Is purchase price over 1M', propertyData['Is purchase price over 1M'])
                    }`}
                  >
                    <option value="">Select...</option>
                    <option value="1M or under">1M or under (+1)</option>
                    <option value="1-1.5M">1-1.5M (0)</option>
                    <option value="1.5M +">1.5M + (-1)</option>
                  </select>
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Is value subject to permits
                  </label>
                  <select
                    value={propertyData['Is value subject to permits'] || ''}
                    onChange={(e) => updateField('Is value subject to permits', e.target.value)}
                    className={`w-full px-4 py-2 border-2 rounded-lg focus:ring-2 focus:ring-indigo-500 ${
                      getSelectColor('Is value subject to permits', propertyData['Is value subject to permits'])
                    }`}
                  >
                    <option value="">Select...</option>
                    <option value="No">No (0)</option>
                    <option value="Approved Permits">Approved Permits (0)</option>
                    <option value="Yes">Yes (-2)</option>
                  </select>
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Is value subject to ADU?
                  </label>
                  <select
                    value={propertyData['Is value subject to ADU?'] || ''}
                    onChange={(e) => updateField('Is value subject to ADU?', e.target.value)}
                    className={`w-full px-4 py-2 border-2 rounded-lg focus:ring-2 focus:ring-indigo-500 ${
                      getSelectColor('Is value subject to ADU?', propertyData['Is value subject to ADU?'])
                    }`}
                  >
                    <option value="">Select...</option>
                    <option value="No">No (0)</option>
                    <option value="Approved Permits">Approved Permits (0)</option>
                    <option value="Yes">Yes (-3)</option>
                  </select>
                </div>

                <div>
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Is wholesale price higher than listing asking price?
                  </label>
                  <select
                    value={propertyData['Is wholesale price higher than listing asking price?'] || ''}
                    onChange={(e) => updateField('Is wholesale price higher than listing asking price?', e.target.value)}
                    className={`w-full px-4 py-2 border-2 rounded-lg focus:ring-2 focus:ring-indigo-500 ${
                      getSelectColor('Is wholesale price higher than listing asking price?', propertyData['Is wholesale price higher than listing asking price?'])
                    }`}
                  >
                    <option value="">Select...</option>
